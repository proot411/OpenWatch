cmake_minimum_required(VERSION 3.21)
project(OpenWatch VERSION 0.11.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network Test Xml)
find_package(PkgConfig REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(vault src/vault.cpp)
target_link_libraries(vault PUBLIC Qt6::Core OpenSSL::Crypto)
target_include_directories(vault PUBLIC src)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
add_library(liveaudio src/audio.cpp)
target_include_directories(liveaudio PUBLIC src)
target_link_libraries(liveaudio PUBLIC Qt6::Core PkgConfig::AV PkgConfig::SDL)
add_library(protocol src/dvrip.cpp)
target_link_libraries(protocol PUBLIC Qt6::Network)
target_include_directories(protocol PUBLIC src)
add_library(discovery src/discovery.cpp src/discoverydialog.cpp)
target_link_libraries(discovery PUBLIC protocol Qt6::Widgets Qt6::Xml)
add_executable(openwatch src/controls.cpp src/main.cpp src/video.cpp src/recorderdialog.cpp)
target_link_libraries(openwatch PRIVATE vault discovery protocol Qt6::Widgets liveaudio)
include(CTest)
if(BUILD_TESTING)
 add_executable(protocol_tests tests/protocol_test.cpp)
 target_link_libraries(protocol_tests PRIVATE protocol Qt6::Test)
 add_test(NAME protocol COMMAND protocol_tests)
endif()
install(TARGETS openwatch RUNTIME DESTINATION bin)
set(CPACK_PACKAGE_CONTACT "OpenWatch contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native experimental DVRIP and RTSP video monitor")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_GENERATOR "TGZ;DEB")
include(CPack)
if(BUILD_TESTING)
 add_executable(video_test tests/video_test.cpp src/video.cpp)
 target_link_libraries(video_test PRIVATE protocol Qt6::Gui liveaudio)
 find_package(Python3 COMPONENTS Interpreter)
 find_program(FFMPEG_EXECUTABLE ffmpeg)
 if(Python3_FOUND AND FFMPEG_EXECUTABLE)
  add_test(NAME video_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:video_test>)
  set_tests_properties(video_integration PROPERTIES TIMEOUT 45)
 endif()
endif()
